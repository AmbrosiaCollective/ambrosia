cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(ambrosia_native_host LANGUAGES CXX)
  include(CTest)
endif()

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  message(FATAL_ERROR "The native PE bootstrap currently requires an x86-64 host")
endif()

add_library(ambrosia_host_support STATIC host.cpp host_tls.cpp host_time.cpp host_stack.cpp host_random.cpp)
target_compile_features(ambrosia_host_support PUBLIC cxx_std_20)
target_include_directories(ambrosia_host_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(ambrosia_pe_core STATIC pe_image.cpp pe_exports.cpp guest_thread.cpp module_execution.cpp module_graph.cpp unwind_table.cpp unwind_program.cpp unwind_epilogue.cpp)
target_link_libraries(ambrosia_pe_core PUBLIC ambrosia_host_support)
add_library(ambrosia_package_provider STATIC package_modules.cpp)
target_compile_features(ambrosia_package_provider PUBLIC cxx_std_20)
target_include_directories(ambrosia_package_provider PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(ambrosia_service_runtime STATIC import_registry.cpp service_module_image.cpp native_modules.cpp event_signal.cpp kernel_objects.cpp kernel_services.cpp registry_keys.cpp registry_services.cpp graphics_memory_policy.cpp graphics_services.cpp memory_services.cpp service_runtime.cpp clock_services.cpp critical_sections.cpp critical_services.cpp fls_slots.cpp fls_services.cpp crt_lifecycle.cpp crt_services.cpp crt_exit.cpp crt_exit_services.cpp crt_streams.cpp crt_startup.cpp crt_startup_services.cpp crt_heap.cpp crt_heap_services.cpp slist_services.cpp loader_services.cpp locale_services.cpp crt_locales.cpp pointer_services.cpp startup_trace.cpp host_fault_trace.cpp)
target_link_libraries(ambrosia_service_runtime PUBLIC ambrosia_pe_core)
add_library(ambrosia_native_command STATIC native_command.cpp)
target_link_libraries(ambrosia_native_command PUBLIC ambrosia_service_runtime ambrosia_package_provider)
add_executable(ambrosia-host main.cpp)
target_link_libraries(ambrosia-host PRIVATE ambrosia_native_command)
if(MINGW)
  target_link_options(ambrosia-host PRIVATE -municode)
endif()
foreach(target ambrosia_host_support ambrosia_pe_core ambrosia_package_provider ambrosia_service_runtime ambrosia_native_command ambrosia-host)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(BUILD_TESTING)
  add_executable(ambrosia_native_tests tests.cpp)
  target_compile_definitions(ambrosia_native_tests PRIVATE AMBROSIA_STANDALONE_TESTS)
  target_link_libraries(ambrosia_native_tests PRIVATE ambrosia_native_command)
  add_test(NAME native_host_bootstrap COMMAND ambrosia_native_tests)
  set_tests_properties(native_host_bootstrap PROPERTIES TIMEOUT 20)
endif()
